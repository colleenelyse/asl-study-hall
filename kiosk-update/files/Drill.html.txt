<script>
(function () {
  var classKey = '';            // '' = the class in session (bell schedule)
  var data = null, lastGood = 0, offset = 0, busy = false;

  function el(id) { return document.getElementById(id); }
  function esc(t) { return String(t == null ? '' : t).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function fill(id, rows, emptyText) {
    el(id).innerHTML = rows.length ? rows.join('') : '<li class="empty">' + esc(emptyText) + '</li>';
  }
  function mins(ms) { return Math.max(0, Math.floor((Date.now() + offset - ms) / 60000)); }

  function draw() {
    if (!data) return;
    var students = data.students || [];
    var passes = data.passes || [];
    var outNames = {};
    passes.forEach(function (p) { outNames[String(p.name || '').toLowerCase()] = true; });

    var here = [], gone = [];
    students.forEach(function (s) {
      var full = String(s.fullName || s.firstName || '');
      if (outNames[full.toLowerCase()]) return;   // listed under "out"
      if (s.kind === 'absent') gone.push(full); else here.push(full);
    });
    here.sort(); gone.sort();

    var out = passes.slice().sort(function (a, b) { return a.timeOutMs - b.timeOutMs; }).map(function (p) {
      return '<li><span>' + esc(p.name) + '</span><small>' + esc(p.destination) + '<br>' + mins(p.timeOutMs) + ' min</small></li>';
    });

    el('nHere').textContent = here.length;
    el('nOut').textContent = passes.length;
    el('nGone').textContent = gone.length;
    fill('outList', out, 'No one is out.');
    fill('goneList', gone.map(function (n) { return '<li>' + esc(n) + '</li>'; }),
      data.selected ? 'Everyone has checked in.' : 'Choose a class above.');
    fill('hereList', here.map(function (n) { return '<li>' + esc(n) + '</li>'; }),
      data.selected ? 'No one has checked in yet.' : 'Choose a class above.');

    var sel = data.selected;
    el('sub').textContent = sel ? (sel.label || (sel.period + ' ' + sel.classTitle)) + (sel.live ? ' · in session now' : '') :
      'No class in session — choose one';
  }

  function drawPicker() {
    var pick = el('pick');
    var html = '<option value="">Class in session now</option>';
    (data.classes || []).forEach(function (c) {
      html += '<option value="' + esc(c.key) + '">' + esc(c.label) + '</option>';
    });
    if (pick.innerHTML !== html) pick.innerHTML = html;
    pick.value = classKey;
  }

  function refresh() {
    if (busy) return;
    busy = true;
    var asked = classKey;
    google.script.run
      .withSuccessHandler(function (result) {
        busy = false;
        if (asked !== classKey) { refresh(); return; }
        if (!result || !result.success) return;
        data = result; lastGood = Date.now();
        offset = Number(result.nowMs || Date.now()) - Date.now();
        drawPicker(); draw(); sync();
      })
      .withFailureHandler(function () { busy = false; sync(); })
      .getTeacherDashboard({ classKey: classKey });
  }

  function sync() {
    el('sync').textContent = lastGood
      ? 'Updated ' + Math.round((Date.now() - lastGood) / 1000) + ' s ago · refreshes every 10 s'
      : 'Connecting…';
  }

  el('pick').addEventListener('change', function () { classKey = this.value; refresh(); });
  setInterval(function () { draw(); sync(); }, 1000);
  setInterval(function () { if (!document.hidden) refresh(); }, 10000);
  refresh();
})();
</script>
