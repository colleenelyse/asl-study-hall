  const begin = p => {
    const tool = String(p.tool || 'timer');
    document.title = { timer: 'Timer', dice: 'Dice', score: 'Scoreboard', rnum: 'Random number', passboard: 'Hall pass board', timeleft: 'Time left' }[tool] || 'Classroom tools';
    document.body.classList.add('ktools');
    if (!VIEW[tool]) return go('timer');
    go(tool);
    if (KTOOLS[tool] && ACT[KTOOLS[tool]]) setTimeout(() => { try { ACT[KTOOLS[tool]]($('[data-a="' + KTOOLS[tool] + '"]') || document.body); } catch (e) {} }, 300);
    if (tool === 'passboard') { pbLoad(); setInterval(pbLoad, 10000); setInterval(() => document.querySelectorAll('[data-out]').forEach(i => { i.textContent = pbClock(+i.dataset.out); }), 1000); }
    if (tool === 'timeleft') { tlLoad(); setInterval(tlLoad, 60000); setInterval(tlPaint, 1000); }
  };
  try { google.script.url.getLocation(loc => begin((loc && loc.parameter) || {})); }
  catch (e) { begin({}); }
})();

</script>
</body>
</html>
