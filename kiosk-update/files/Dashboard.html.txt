  function chime() {
    if (!soundOn) return;
    try {
      audioCtx = audioCtx || new (window.AudioContext || window.webkitAudioContext)();
      [0, 0.22].forEach(function (at, i) {
        var o = audioCtx.createOscillator(), g = audioCtx.createGain();
        o.type = 'sine'; o.frequency.value = i ? 660 : 880;
        g.gain.setValueAtTime(0.0001, audioCtx.currentTime + at);
        g.gain.exponentialRampToValueAtTime(0.25, audioCtx.currentTime + at + 0.02);
        g.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + at + 0.35);
        o.connect(g); g.connect(audioCtx.destination);
        o.start(audioCtx.currentTime + at); o.stop(audioCtx.currentTime + at + 0.4);
      });
    } catch (e) {}
  }

      .withSuccessHandler(function (result) {

        failures = 0; lastGood = Date.now();
        offset = Number(result.nowMs || Date.now()) - Date.now();
        data = result;
        drawAll(); drawSync();
